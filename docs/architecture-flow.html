flowchart TD
    U["USER REQUEST<br/>[DATA · renderer state]<br/>in: prompt, files, persona, model, enabled tools<br/>out: typed run request"] --> P{"PERMISSION GATE<br/>[FUNCTION]<br/>condition: requested files, hosts, and tools approved?"}
    P -->|"no"| ASK["PERMISSION PROMPT<br/>[FUNCTION · no LLM writes it]<br/>in: missing grants<br/>out: user decision"]
    ASK -. "reply carried as run state" .-> P
    P -->|"yes"| C["CONTEXT ASSEMBLER<br/>[FUNCTION]<br/>in: builder prompt, checkpoint, recent turns, retrieved wiki pages, attachments<br/>out: stable provider context"]
    C --> L["BUILDER TURN<br/>[AGENT · user-selected model]<br/>in: assembled context + tool schemas<br/>out: answer or tool calls"]
    L --> T{"TOOL ROUTER<br/>[FUNCTION]<br/>condition: validated tool call requested?"}
    T -->|"yes"| X["LOCAL TOOL EXECUTION<br/>[LIBRARY · Electron main / Pyodide]<br/>in: schema-valid call + grants<br/>out: bounded result + trace"]
    X --> L
    T -->|"no"| R{"REVIEW MODE<br/>[FUNCTION]<br/>condition: reviewer persona enabled?"}
    R -->|"yes"| V["REVIEWER TURN<br/>[AGENT · user-selected model]<br/>in: original ask + answer + reviewer prompt<br/>out: critique"]
    V --> Q["REVISION TURN<br/>[AGENT · builder model]<br/>in: answer + critique + binding format contract<br/>out: revised answer"]
    R -->|"no"| O["VISIBLE RESPONSE<br/>[DATA · local conversation]<br/>out: answer + activity trace"]
    Q --> O
    O --> M{"COMPACTION GATE<br/>[FUNCTION]<br/>condition: token/turn threshold or session close?"}
    M -->|"not reached"| U
    M -->|"reached"| K["MEMORY COMPILER<br/>[AGENT + deterministic verifier]<br/>in: immutable raw evidence + previous checkpoint<br/>out: checkpoint + proposed wiki changes"]
    K --> W["LOCAL MEMORY<br/>[DATA · Markdown source + rebuildable SQLite FTS]<br/>out: conversational and long-term retrieval"]
    W --> U
    E["EXTERNAL CONNECTOR BUILDER<br/>[AGENT + FUNCTION validator]<br/>in: uploaded API docs + user-approved auth fields<br/>out: non-secret manifest + runtime tools"] --> T
    S["SESSION SECRET VAULT<br/>[DATA · volatile memory only]<br/>in: provider and connector credentials<br/>out: request-scoped auth; cleared on quit"] --> L
    S --> X
